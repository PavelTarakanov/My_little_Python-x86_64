section .data
	n dq 0
	input_str times 20 db 0
	output_str times 20 db 0
section .text
	global _start
_start:
	mov rax, [n]
	mov rcx, 2
	sub rax, rcx
	cmp rax, 0
	je label_0
	mov rcx, 123213
	mov [n], rcx

label_0:

	mov rax, 60
	xor rdi, rdi
	syscall
;------------------------------------------------------------------
print:
	lea rdi, output_str

	cmp rax, 0
	jg pozitive
	not rax
	inc rax
	mov byte [rdi], '-'
	inc rdi
pozitive:
	xor rcx, rcx                ;rcx = 0
	mov cl, 0                   ;counter

dec_count_digits:
	inc cl                      ;cl++
	xor rdx, rdx
	mov rbx, 10
	div rbx
	push rdx
	test rax, rax
	jnz dec_count_digits

print_dec:
	pop rax
	add al, '0'
	mov [rdi], al
	inc rdi                     ;кdi++
	loop print_dec

	mov byte [rdi], 10

print_and_free_buffer:
	lea rsi, output_str      ; copy address
	lea rdi, output_str      ; copy begin(to cmp symbol abd '\0')
	xor rcx, rcx            ; len counter

len_calculate:
	cmp byte [rdi], 0       ; if '\0'
	je print_str
	inc rcx                 ; len counter++
	inc rdi                 ; next symabol
	jmp len_calculate

print_str:
	mov rax, 1              ; syscall number
	mov rdi, 1              ; stdout
	mov rdx, rcx            ; str len
	syscall                 ; print str

	mov rcx, 20
	lea rdi, output_str

free_buffer:

	mov byte [rdi], 0
	inc rdi
	loop free_buffer

	ret
;------------------------------------------------------------
input:
	push rbx
	push rcx
	push rdx
	push rdi
	push rsi
	mov rax, 0
	mov rdi, 0
	lea rsi, input_str
	mov rdx, 20
	syscall

	lea rdi, input_str
	xor rax, rax
	mov rsi, 1                   ;флаг знака

	xor rcx, rcx
	skip_spaces:
	mov cl, [rdi]
	cmp cl, ' '
	jne check_sign
	inc rdi
	jmp skip_spaces
check_sign:
	mov cl, [rdi]
	cmp cl, '-'
	je set_negative
	jmp make_number
set_negative:
	mov rsi, -1
	inc rdi

make_number:
	mov cl, [rdi]
	cmp cl, '0'
	jl done
	cmp cl, '9'
	jg done

	sub cl, '0'
	imul rax, 10
	add rax, rcx

	inc rdi
	jmp make_number
done:
	imul rax, rsi
	mov rcx, 20
	lea rdi, input_str
free_input_buffer:
	mov byte [rdi], 0
	inc rdi
	loop free_input_buffer

	pop rsi
	pop rdi
	pop rdx
	pop rcx
	pop rbx

	ret
